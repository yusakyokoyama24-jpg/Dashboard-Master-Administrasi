    // Render Leger Nilai: 10 Sumatif (80%) + SAS (20%) + Korelasi Kehadiran
    function renderLegerNilai() {
      const tbody = document.getElementById('leger-table-body');
      if (!tbody) return;
      tbody.innerHTML = '';

      currentSiswaList.forEach((s, idx) => {
        const pres = currentPresensi[s.id] || { h: 18, pct: 100 };
        const studentGrades = currentNilaiList.filter(n => n.siswaId === s.id);
        const sumatifGrades = studentGrades.filter(n => n.jenis === 'sumatif');
        const pasGrade = studentGrades.find(n => n.jenis === 'pas');

        // Nilai Sumatif 1 s/d 10
        const sVals = [];
        for (let i = 1; i <= 10; i++) {
          const found = sumatifGrades.find(n => n.sumatifKe === i);
          sVals.push(found ? found.skor : null);
        }

        const filledScores = sVals.filter(v => v !== null);
        const avgSumatif = filledScores.length > 0
          ? Math.round(filledScores.reduce((a, b) => a + b, 0) / filledScores.length)
          : 0;
        const sasSkor = pasGrade ? pasGrade.skor : 0;

        // Formula Nilai Akhir: (Rata Sumatif × 80%) + (SAS × 20%)
        const na = avgSumatif > 0
          ? (sasSkor > 0 ? Math.round(avgSumatif * 0.8 + sasSkor * 0.2) : avgSumatif)
          : (sasSkor > 0 ? sasSkor : 0);

        let predikat = 'D';
        if (na >= 90) predikat = 'A';
        else if (na >= 80) predikat = 'B';
        else if (na >= 70) predikat = 'C';

        // Korelasi Indicator
        let korelasiBadge = '<span class="text-blue-600 font-bold">Stabil</span>';
        if (pres.pct >= 85 && avgSumatif >= 80) {
          korelasiBadge = '<span class="text-emerald-600 font-bold">Optimal</span>';
        } else if (pres.pct < 80 && avgSumatif < 75) {
          korelasiBadge = '<span class="text-rose-600 font-bold">At-Risk</span>';
        }

        const deskripsi = na >= 80
          ? 'Menunjukkan penguasaan materi sumatif sangat baik dan konsisten.'
          : 'Perlu bimbingan dan pendampingan konsep pada asesmen remedial.';

        const tr = document.createElement('tr');
        tr.className = 'hover:bg-slate-50 transition text-slate-800';
        tr.innerHTML = `
          <td class="py-2 px-1 text-center font-bold">${idx + 1}</td>
          <td class="py-2 px-2 text-center font-mono">${s.nisn}</td>
          <td class="py-2 px-2 font-medium">${s.nama}</td>
          <td class="py-2 px-1 text-center font-bold text-teal-700 bg-teal-50/50">${pres.pct}%</td>
          <td class="py-2 px-1 text-center text-[10px] no-print">${korelasiBadge}</td>
          ${sVals.map(v => `<td class="py-1 px-1 text-center border-l border-slate-100">${v !== null ? v : '-'}</td>`).join('')}
          <td class="py-2 px-1 text-center font-bold text-amber-700 bg-amber-50/50 border-l border-slate-200">${avgSumatif || '-'}</td>
          <td class="py-2 px-1 text-center font-semibold text-slate-700">${sasSkor || '-'}</td>
          <td class="py-2 px-1 text-center font-black text-blue-900 bg-blue-50/50 text-sm">${na || '-'}</td>
          <td class="py-2 px-1 text-center font-bold">${predikat}</td>
          <td class="py-2 px-2 text-[10px] text-slate-600 truncate max-w-[150px]" title="${deskripsi}">${deskripsi}</td>
          <td class="py-2 px-1 text-center no-print">
            <button onclick="editNilaiSiswa('${s.id}')" class="text-indigo-600 hover:text-indigo-900 font-semibold text-xs">Edit</button>
          </td>
        `;
        tbody.appendChild(tr);
      });
    }

    function printLaporanNilai() {

              size: A4 landscape;
              margin: 6mm !important;
            }
          }
        `;
        document.head.appendChild(style);
      }
      window.print();
    }

    // Sinkronisasi dengan Google Spreadsheet / Apps Script
    function sinkronisasiData() {
      if (typeof google !== 'undefined' && google.script && google.script.run) {
        Swal.fire({
          title: 'Menghubungkan ke Sheets...',
          text: 'Mengambil data siswa, absensi, dan leger nilai terbaru.',
          allowOutsideClick: false,
          didOpen: () => { Swal.showLoading(); }
        });

        google.script.run
          .withSuccessHandler(function(response) {
            Swal.close();
            if (response && response.status === 'success') {
              if (response.siswa) currentSiswaList = response.siswa;
              if (response.nilai) currentNilaiList = response.nilai;
              renderSiswa();
              renderLegerNilai();
              Swal.fire('Sinkronisasi Sukses', 'Data berhasil diperbarui dari Google Spreadsheet!', 'success');
            }
          })
          .withFailureHandler(function(error) {
            Swal.close();
            Swal.fire('Gagal Sinkron', error.toString(), 'error');
          })
          .apiGetAllData();
      } else {
        Swal.fire({
          title: 'Data Lokal Aktif',
          text: 'Aplikasi berjalan dalam mode simulasi web. Data tersimpan di memori peramban.',
          icon: 'info',
          timer: 2000,
          showConfirmButton: false
        });
      }
    }

    function bukaModalTambahSiswa() {
      Swal.fire({
        title: 'Tambah Siswa Baru',
        html: `
          <input id="swal-nisn" class="swal2-input" placeholder="NISN Siswa">
          <input id="swal-nama" class="swal2-input" placeholder="Nama Lengkap Siswa">
          <input id="swal-kelas" class="swal2-input" placeholder="Kelas (misal: X RPL 1)" value="X RPL 1">
        `,
        focusConfirm: false,
        preConfirm: () => {
          return {
            nisn: document.getElementById('swal-nisn').value,
            nama: document.getElementById('swal-nama').value,
            kelas: document.getElementById('swal-kelas').value
          };
        }
      }).then((res) => {
        if (res.isConfirmed && res.value.nama) {
          const newS = {
            id: 's_' + Date.now(),
            nisn: res.value.nisn || '00' + Math.floor(10000000 + Math.random() * 90000000),
            nama: res.value.nama,
            kelas: res.value.kelas,
            jk: 'L'
          };
          currentSiswaList.push(newS);
          renderSiswa();
          renderLegerNilai();
          Swal.fire('Siswa Ditambahkan', `Siswa ${newS.nama} berhasil didaftarkan.`, 'success');
        }
      });
    }

    function bukaModalEntriNilai() {
      Swal.fire({
        title: 'Entri Nilai Sumatif',
        html: `
          <select id="swal-siswa" class="swal2-input">
            ${currentSiswaList.map(s => `<option value="${s.id}">${s.nama} (${s.kelas})</option>`).join('')}
          </select>
          <select id="swal-ke" class="swal2-input">
            ${Array.from({length: 10}, (_, i) => `<option value="${i+1}">Sumatif ${i+1}</option>`).join('')}
            <option value="pas">Sumatif Akhir Semester (SAS)</option>
          </select>
          <input id="swal-skor" type="number" class="swal2-input" placeholder="Skor Nilai (0 - 100)" min="0" max="100" value="85">
        `,
        preConfirm: () => {
          return {
            siswaId: document.getElementById('swal-siswa').value,
            ke: document.getElementById('swal-ke').value,
            skor: Number(document.getElementById('swal-skor').value)
          };
        }
      }).then((res) => {
        if (res.isConfirmed) {
          if (res.value.ke === 'pas') {
            currentNilaiList.push({
              siswaId: res.value.siswaId,
              jenis: 'pas',
              skor: res.value.skor
            });
          } else {
            currentNilaiList.push({
              siswaId: res.value.siswaId,
              jenis: 'sumatif',
              sumatifKe: Number(res.value.ke),
              skor: res.value.skor
            });
          }
          renderLegerNilai();
          Swal.fire('Nilai Disimpan', 'Nilai berhasil dicatat ke dalam leger.', 'success');
        }
      });
    }
